// Drishtec - Phase 1 (cap-mounted): detect obstacles from 5 cm to 40 cm
// Closer obstacle = faster beeps. Outside 5-40 cm = silence.
//
// Wiring (same as phase-1, ESP32):
//   HC-SR04 VCC  -> 5V (VIN)        HC-SR04 GND -> GND
//   HC-SR04 TRIG -> GPIO 5
//   HC-SR04 ECHO -> GPIO 18 (through voltage divider: 1k series + 2k to GND)
//   Buzzer (+)   -> GPIO 19         Buzzer (-)  -> GND
//
// Needs arduino-esp32 core 3.x (tone() support).

const int TRIG_PIN   = 5;
const int ECHO_PIN   = 18;
const int BUZZER_PIN = 19;

const float MIN_CM = 5.0;      // closest distance we care about
const float MAX_CM = 40.0;     // farthest distance we care about

const int BEEP_FREQ_HZ   = 1000;
const int BEEP_LENGTH_MS = 40;
const int FASTEST_GAP_MS = 80;    // gap between beeps at MIN_CM
const int SLOWEST_GAP_MS = 600;   // gap between beeps at MAX_CM

const unsigned long PING_EVERY_MS = 40;   // enough time for echoes to die out
const unsigned long ECHO_TIMEOUT_US = 3500; // ~60 cm round trip, keeps loop fast

const int CONFIRM_HITS = 2;   // readings in range needed before beeping
                              // (filters single false echoes from head movement)

unsigned long lastPing = 0;
unsigned long lastBeep = 0;
float distanceCm = -1;   // -1 means nothing detected in range
int hitCount = 0;

float readDistanceCm() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  unsigned long duration = pulseIn(ECHO_PIN, HIGH, ECHO_TIMEOUT_US);
  if (duration == 0) return -1;            // no echo

  float cm = duration * 0.0343 / 2.0;
  if (cm < MIN_CM || cm > MAX_CM) return -1;
  return cm;
}

void setup() {
  Serial.begin(115200);
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(BUZZER_PIN, OUTPUT);
}

void loop() {
  unsigned long now = millis();

  // 1. Take a new distance reading on schedule
  if (now - lastPing >= PING_EVERY_MS) {
    lastPing = now;
    float reading = readDistanceCm();

    if (reading > 0) {
      if (hitCount < CONFIRM_HITS) hitCount++;
      if (hitCount >= CONFIRM_HITS) distanceCm = reading;
    } else {
      hitCount = 0;
      distanceCm = -1;
    }
    Serial.println(distanceCm);
  }

  // 2. Beep faster as the obstacle gets closer (no blocking delay)
  if (distanceCm > 0) {
    float t = (distanceCm - MIN_CM) / (MAX_CM - MIN_CM);   // 0 = near, 1 = far
    long gap = FASTEST_GAP_MS + (long)(t * (SLOWEST_GAP_MS - FASTEST_GAP_MS));
    gap = constrain(gap, FASTEST_GAP_MS, SLOWEST_GAP_MS);

    if (now - lastBeep >= (unsigned long)gap) {
      lastBeep = now;
      tone(BUZZER_PIN, BEEP_FREQ_HZ, BEEP_LENGTH_MS);
    }
  }
}
